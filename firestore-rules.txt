rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Qui est qui ──
    function signedIn() {
      return request.auth != null && request.auth.token.email_verified == true;
    }
    function isOwner() {
      // Deux adresses propriétaires : l'adresse de l'activité, et l'adresse
      // personnelle en secours si la première venait à être désactivée.
      return signedIn() && request.auth.token.email.lower() in
             ['dollars.sourcing@gmail.com', 'chweirtzleryoyo@gmail.com'];
    }
    // Un invité est autorisé si son email figure dans settings/access
    function isTeam() {
      return isOwner() || (
        signedIn()
        && exists(/databases/$(database)/documents/settings/access)
        && request.auth.token.email in
             get(/databases/$(database)/documents/settings/access).data.emails
      );
    }
    function isMine() {
      return signedIn() && resource.data.ownerId == request.auth.uid;
    }
    function claimingMine() {
      return signedIn() && request.resource.data.ownerId == request.auth.uid;
    }

    // ── CLIENTS ──
    // Chacun ne voit que les siens dans l'app.
    // La lecture d'une fiche isolée reste ouverte pour le portail client
    // (le lien exige id + token, et aucune donnée financière n'y figure).
    match /clients/{clientId} {
      allow get: if true;
      allow list: if isTeam();
      allow create: if isTeam() && claimingMine();
      allow update, delete: if isTeam() && isMine();
    }

    // ── COMMANDES ── fermées, prix d'achat et marge compris
    // Le portail client ne les lit plus : il lit `publicOrders` ci-dessous.
    match /orders/{orderId} {
      allow get, list: if isTeam();
      allow create: if isTeam() && claimingMine();
      allow update, delete: if isTeam() && isMine();
    }

    // ── COPIE PUBLIQUE DES COMMANDES ──
    // Le portail client n'est pas connecté : Firestore le voit comme un visiteur
    // anonyme, et sa requête (« toutes celles dont clientId vaut le mien »)
    // relève de `list`, jamais de `get`. Il FAUT donc que `list` soit ouvert ici
    // — c'est le piège qui avait donné un portail vide sans aucune erreur.
    //
    // Ce qu'on ouvre, en revanche, ne contient QUE ce que le client voit déjà :
    // produit, quantité, statut, prix client, acompte, expédition, photo.
    // Ni prix d'achat, ni marge. C'est l'app qui écrit cette copie, et seul le
    // propriétaire d'une commande peut toucher la sienne.
    match /publicOrders/{orderId} {
      allow get, list: if true;
      allow create: if isTeam() && claimingMine();
      allow update, delete: if isTeam() && isMine();
    }

    // ── DEMANDES CLIENTS ──
    // Déposées sans compte depuis le portail, traitées seulement par leur
    // destinataire. Même remarque que ci-dessus : le portail liste les siennes.
    // La création reste validée, c'est la seule porte d'écriture publique.
    match /pendingOrders/{reqId} {
      allow get: if true;
      allow list: if true;
      allow create: if request.resource.data.product is string
                    && request.resource.data.product.size() > 0
                    && request.resource.data.product.size() < 200
                    && request.resource.data.qty is int
                    && request.resource.data.qty > 0
                    && request.resource.data.qty < 100000;
      allow update, delete: if isTeam() && isMine();
    }

    // ── FOURNISSEURS ── partagés par l'équipe
    // Tout le monde lit et modifie ; seul le créateur (ou le propriétaire) supprime.
    match /suppliers/{supplierId} {
      allow read: if isTeam();
      allow create: if isTeam() && claimingMine();
      allow update: if isTeam();
      allow delete: if isOwner() || isMine();
    }

    // ── RÉGLAGES ──
    // La liste des accès : lisible par tout compte connecté (nécessaire pour qu'un
    // invité vérifie sa propre autorisation), modifiable uniquement par le propriétaire.
    match /settings/access {
      allow read: if signedIn();
      allow write: if isOwner();
    }
    // Mémoire de l'assistant : propriétaire uniquement.
    match /settings/{docId} {
      allow read, write: if isOwner();
    }

    // ── RÉGLAGES PERSONNELS ──
    // Le code PIN de chacun, dans son propre espace : lui seul y accède.
    match /userSettings/{uid} {
      allow read, write: if signedIn() && request.auth.uid == uid;
    }

    // ── Tout le reste : interdit ──
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

// ─────────────────────────────────────────────────────────────────────────
// NOTE — pourquoi `list` est ouvert sur publicOrders et pendingOrders
//
// Le portail client n'est jamais connecté à Firebase : il s'identifie avec son
// lien (id + token) et son code à 6 chiffres, vérifiés dans le navigateur.
// Firestore ne peut donc pas savoir « qui » demande, et une requête filtrée sur
// clientId reste, de son point de vue, une requête anonyme sur la collection.
// C'est pour ça que `list` doit rester ouvert quelque part.
//
// Ce qui a changé : ce n'est plus `orders` qui est ouvert, mais `publicOrders`,
// une copie qui ne contient que ce que le client voit déjà à l'écran. Le prix
// d'achat et la marge ne quittent plus `orders`, désormais fermé à l'équipe.
//
// Ce qui reste ouvert et qu'il faut connaître : `pendingOrders`. Un visiteur
// anonyme peut lister les demandes — un nom de client et un produit, jamais un
// montant. Le refermer demanderait de faire transiter les demandes par une
// fonction serveur ; ce n'est pas justifié tant qu'elles ne portent pas d'argent.
//
// `clients` n'a jamais été exposé à la liste : seul `get` est ouvert, et il
// exige l'identifiant exact du document — celui qui est dans le lien du client.
// ─────────────────────────────────────────────────────────────────────────
